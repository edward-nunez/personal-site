📋 Getting Started:

1. Check deployment status:
   kubectl get pods -n {{ .Release.Namespace }}

2. View services:
   kubectl get services -n {{ .Release.Namespace }}

{{- if .Values.ingress.enabled }}

3. Access your application:
   {{- range .Values.ingress.hosts }}
   https://{{ .host }}
   {{- end }}

{{- else if eq .Values.frontend.service.type "LoadBalancer" }}

3. Get the LoadBalancer IP:
   kubectl get service {{ include "personal-site.frontend.serviceName" . }} -n {{ .Release.Namespace }}
   
   Then access: http://<EXTERNAL-IP>

{{- else }}

3. Port-forward to access locally:
   kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "personal-site.frontend.serviceName" . }} 8080:{{ .Values.frontend.service.port }}
   
   Then access: http://localhost:8080

{{- end }}
